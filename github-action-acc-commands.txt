# (This fixes the 'getAccessToken' error for docker auth)
gcloud iam service-accounts add-iam-policy-binding $DEFAULT_SA \
    --member="serviceAccount:${DEFAULT_SA}" \
    --role="roles/iam.serviceAccountTokenCreator"

# === COMMAND 3: GRANT ARTIFACT REGISTRY WRITER ROLE ===
# (This fixes the 'uploadArtifacts' error for docker push)

    --location=us-central1 \
    --member="serviceAccount:${DEFAULT_SA}" \
    --role="roles/artifactregistry.writer"
