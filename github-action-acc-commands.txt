------------------------------------------------------------------------
gcloud iam service-accounts create github-actions-sa --display-name="GitHub Actions Deployer"
------------------------------------------------------------------
export PROJECT_ID=$(gcloud config get-value project)
---------------------------------------------------------------
# Allow pushing to Artifact Registry
gcloud artifacts repositories add-iam-policy-binding iris-app-repo \
    --location=us-central1 \
    --member="serviceAccount:github-actions-sa@${PROJECT_ID}.iam.gserviceaccount.com" \
    --role="roles/artifactregistry.writer"
-------------------------------------------------------------------
# Allow deploying to GKE
gcloud projects add-iam-policy-binding $PROJECT_ID \
    --member="serviceAccount:github-actions-sa@${PROJECT_ID}.iam.gserviceaccount.com" \
    --role="roles/container.developer"
----------------------------------------------------------------
# 1. Enable the necessary API
gcloud services enable iamcredentials.googleapis.com
----------------------------------------------------------------
# 2. Create the Workload Identity Pool
gcloud iam workload-identity-pools create "github-pool" \
    --project=$PROJECT_ID \
    --location=global
    --display-name="GitHub Actions Pool"
----------------------------------------------------------------------
# 3. Get the ID of the new pool
export POOL_ID=$(gcloud iam workload-identity-pools describe "github-pool" \
    --project=$PROJECT_ID --format='value(name)')
    
Result : projects/1081441859341/locations/global/workloadIdentityPools/github-pool

-------------------------------------------------------------------------
------------------------------------------------------------------------------------------
# 4. Create a provider in that pool for GitHub
gcloud iam workload-identity-pools providers create-oidc "github-provider" \
    --project=$PROJECT_ID \
    --workload-identity-pool="github-pool" \
    --issuer-uri="https://token.actions.githubusercontent.com" \
    --attribute-mapping="google.subject=assertion.sub"

----------------------------------------------------------------
-------------------------------------------------------------------------

# 5. Bind your GSA to your GitHub repo
----------------------------------------------
export GITHUB_REPO="Ripu0902/MLops-Week4"
--------------------------------------
gcloud iam service-accounts add-iam-policy-binding "github-actions-sa@${PROJECT_ID}.iam.gserviceaccount.com" \
    --project=$PROJECT_ID \
    --role="roles/iam.workloadIdentityUser" \
    --member="principalSet://iam.googleapis.com/$POOL_ID/attribute.assertion.sub/repo:${GITHUB_REPO}:ref:refs/heads/main"
-------------------------------------
gcloud iam service-accounts add-iam-policy-binding "github-actions-sa@${PROJECT_ID}.iam.gserviceaccount.com" \
    --project=$PROJECT_ID \
    --role="roles/iam.workloadIdentityUser" \
    --member="principalSet://iam.googleapis.com/$POOL_ID/attribute.assertion.sub/repo:${GITHUB_REPO}:ref:refs/heads/week6"
